package com.torto.myapplication.presentation


import android.Manifest
import android.content.Context
import android.content.pm.PackageManager
import android.hardware.Sensor
import android.hardware.SensorEvent
import android.hardware.SensorEventListener
import android.hardware.SensorManager
import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.activity.result.contract.ActivityResultContracts
import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.runtime.Composable
import androidx.compose.runtime.mutableStateOf
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.text.style.TextAlign
import androidx.compose.ui.unit.sp
import androidx.core.content.ContextCompat
import androidx.wear.compose.material.Button
import androidx.wear.compose.material.MaterialTheme
import androidx.wear.compose.material.Text

class MainActivity : ComponentActivity() {

    private val permissionState = mutableStateOf("Verifica permesso in corso...")
    private val heartRateState = mutableStateOf("Attesa dati...")

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        // Launcher per richiedere il permesso
        val requestPermissionLauncher = registerForActivityResult(
            ActivityResultContracts.RequestPermission()
        ) { isGranted: Boolean ->
            updatePermissionStatus(isGranted)
        }

        // Imposta l'interfaccia con Jetpack Compose
        setContent {
            WearApp(
                permissionStatus = permissionState.value,
                heartRate = heartRateState.value,
                onRequestPermission = {
                    requestPermissionLauncher.launch("android.permission.health.READ_HEART_RATE")
                }
            )
        }

        // Verifica il permesso all'avvio
        checkAndRequestPermission(requestPermissionLauncher)
    }

    override fun onResume() {
        super.onResume()
        // Verifica nuovamente il permesso quando l'app torna in primo piano
        if (ContextCompat.checkSelfPermission(
                this,
                Manifest.permission.BODY_SENSORS
            ) == PackageManager.PERMISSION_GRANTED
        ) {
            updatePermissionStatus(true)
        }
    }

    private fun checkAndRequestPermission(
        requestPermissionLauncher: androidx.activity.result.ActivityResultLauncher<String>
    ) {
        when {
            ContextCompat.checkSelfPermission(
                this,
                Manifest.permission.BODY_SENSORS
            ) == PackageManager.PERMISSION_GRANTED -> {
                updatePermissionStatus(true)
            }
            else -> {
                permissionState.value = "Permesso BODY_SENSORS non concesso."
                requestPermissionLauncher.launch("android.permission.health.READ_HEART_RATE")
            }
        }
    }

    private fun updatePermissionStatus(isGranted: Boolean) {
        if (isGranted) {
            permissionState.value = "Permesso BODY_SENSORS concesso!"
            startHeartRateSensor()
        } else {
            permissionState.value = "Permesso BODY_SENSORS negato."
            heartRateState.value = "Permesso negato, sensore non disponibile."
        }
    }

    private fun startHeartRateSensor() {
        val sensorManager = getSystemService(Context.SENSOR_SERVICE) as SensorManager
        val heartRateSensor = sensorManager.getDefaultSensor(Sensor.TYPE_HEART_RATE)

        if (heartRateSensor != null) {
            // Registra il listener per il sensore di frequenza cardiaca
            sensorManager.registerListener(
                object : SensorEventListener {
                    override fun onSensorChanged(event: SensorEvent?) {
                        event?.let {
                            if (it.sensor.type == Sensor.TYPE_HEART_RATE) {
                                val heartRate = it.values[0].toInt()
                                heartRateState.value = "$heartRate BPM"
                            }
                        }
                    }

                    override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {
                        // Non necessario per questo esempio
                    }
                },
                heartRateSensor,
                SensorManager.SENSOR_DELAY_NORMAL
            )
        } else {
            heartRateState.value = "Sensore di frequenza cardiaca non disponibile."
        }
    }
}

@Composable
fun WearApp(
    permissionStatus: String,
    heartRate: String,
    onRequestPermission: () -> Unit
) {
    MaterialTheme {
        Column(
            modifier = Modifier.fillMaxSize(),
            verticalArrangement = Arrangement.Center,
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Text(
                text = "Stato Permesso:\n$permissionStatus",
                textAlign = TextAlign.Center,
                fontSize = 12.sp
            )
            Text(
                text = "Frequenza Cardiaca:\n$heartRate",
                textAlign = TextAlign.Center,
                fontSize = 16.sp,
                modifier = Modifier.weight(1f)
            )
            if (permissionStatus.contains("negato", ignoreCase = true)) {
                Button(onClick = onRequestPermission) {
                    Text("Richiedi Permesso")
                }
            }
        }
    }
}
