TABLE public.exercise_sets (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  exercise_id uuid,
  set_number integer NOT NULL,
  weight_kg numeric,
  reps integer,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT exercise_sets_pkey PRIMARY KEY (id),
  CONSTRAINT exercise_sets_exercise_id_fkey FOREIGN KEY (exercise_id) REFERENCES public.exercises(id)
);
TABLE public.exercises (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  workout_card_id uuid,
  name character varying NOT NULL,
  order_index integer NOT NULL,
  created_at timestamp with time zone DEFAULT now(),
  exercise_id integer,
  CONSTRAINT exercises_pkey PRIMARY KEY (id),
  CONSTRAINT exercises_workout_card_id_fkey FOREIGN KEY (workout_card_id) REFERENCES public.workout_cards(id)
);
TABLE public.starred_exercises (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  exercise_id integer NOT NULL,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT starred_exercises_pkey PRIMARY KEY (id),
  CONSTRAINT starred_exercises_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id)
);
TABLE public.workout_cards (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  name character varying NOT NULL,
  description text,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT workout_cards_pkey PRIMARY KEY (id),
  CONSTRAINT workout_cards_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id)
);
TABLE public.workout_session_exercises (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  workout_session_id uuid NOT NULL,
  original_exercise_id uuid,
  name character varying NOT NULL,
  order_index integer NOT NULL,
  is_completed boolean NOT NULL DEFAULT false,
  start_time timestamp with time zone,
  end_time timestamp with time zone,
  rest_time_between_sets bigint NOT NULL DEFAULT 60000,
  notes text,
  CONSTRAINT workout_session_exercises_pkey PRIMARY KEY (id),
  CONSTRAINT workout_session_exercises_original_exercise_id_fkey FOREIGN KEY (original_exercise_id) REFERENCES public.exercises(id),
  CONSTRAINT workout_session_exercises_workout_session_id_fkey FOREIGN KEY (workout_session_id) REFERENCES public.workout_sessions(id)
);
TABLE public.workout_session_sets (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  exercise_id uuid NOT NULL,
  set_number integer NOT NULL,
  target_weight_kg numeric,
  actual_weight_kg numeric,
  target_reps integer,
  actual_reps integer,
  is_completed boolean NOT NULL DEFAULT false,
  completed_at timestamp with time zone,
  rest_time_ms bigint,
  rpe integer CHECK (rpe >= 1 AND rpe <= 10),
  notes text,
  CONSTRAINT workout_session_sets_pkey PRIMARY KEY (id),
  CONSTRAINT workout_session_sets_exercise_id_fkey FOREIGN KEY (exercise_id) REFERENCES public.workout_session_exercises(id)
);
TABLE public.workout_sessions (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id uuid,
  workout_card_id uuid,
  name character varying NOT NULL,
  start_time timestamp with time zone NOT NULL,
  end_time timestamp with time zone,
  is_completed boolean NOT NULL DEFAULT false,
  total_duration_ms bigint NOT NULL DEFAULT 0,
  notes text,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT workout_sessions_pkey PRIMARY KEY (id),
  CONSTRAINT workout_sessions_workout_card_id_fkey FOREIGN KEY (workout_card_id) REFERENCES public.workout_cards(id),
  CONSTRAINT workout_sessions_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id)
);
